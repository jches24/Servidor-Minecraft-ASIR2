# ─────────────────────────────────────────────────────────
# crontab.txt
# Tareas programadas del servidor Minecraft Analytics
# Proyecto: Minecraft Analytics Platform - ASIR2 2025-2026
#
# Para aplicar: ejecutar 'crontab -e' y pegar estas líneas
# ─────────────────────────────────────────────────────────

# Backup automático cada día a las 4:00 AM
0 4 * * * /bin/bash /scripts/backup.sh >> /var/log/minecraft-backup.log 2>&1

# Reiniciar data-collector si el contenedor se detiene (cada 5 minutos)
*/5 * * * * docker ps | grep -q data-collector || docker start data-collector

# Limpiar logs antiguos del sistema cada domingo a las 3:00 AM
0 3 * * 0 find /var/log -name "*.log" -mtime +30 -delete
